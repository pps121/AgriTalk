% SLIDE 1 — The Real Problem
% =====================================================================
\begin{frame}{The Real Problem Is Not ``Build a Chatbot for Farmers''}

\begin{columns}[T]
\column{0.52\textwidth}
\begin{block}{A Field Operator Says\ldots}
\begin{center}
\Large\itshape
``Spray more pesticide on\\
the struggling section\\
near the east boundary.''
\end{center}
\end{block}

\medskip
\small This single utterance is simultaneously ambiguous over:
\begin{itemize}
  \item \textbf{Spatial}: which section? (NDVI map required)
  \item \textbf{Quantitative}: ``more'' relative to what baseline?
  \item \textbf{Chemical}: which pesticide, which dosage?
  \item \textbf{Temporal}: safe \emph{now}? (current wind speed?)
\end{itemize}

\column{0.46\textwidth}
\begin{alertblock}{Consequences of Getting It Wrong}
\begin{itemize}
  \item Wrong zone $\to$ crop destruction
  \item Wrong chemical $\to$ resistance development
  \item Wrong conditions $\to$ environmental harm + regulatory violation
\end{itemize}
\end{alertblock}

\medskip
\begin{exampleblock}{And of Over-Correction}
\small A system that asks for confirmation on every query $\to$
\textbf{alert fatigue $\to$ system abandonment}
\end{exampleblock}

\end{columns}

\medskip
\begin{center}
\textbf{\color{techblue}Research challenge: resolve ambiguity with statistically grounded confidence,
real-time field context, and explanations operators can audit and trust.}
\end{center}

\end{frame}

% SLIDE 2 — Research Gap: 3 Challenges
% =====================================================================
\begin{frame}{Three Interlocking Scientific Challenges}

\begin{columns}[T]
\column{0.32\textwidth}
\begin{block}{\#1 Calibration Gap}
LLM softmax is overconfident on OOD agricultural inputs (Guo et al., ICML 2017).\\[4pt]
\textit{Need: statistically valid uncertainty, not heuristic thresholds.}
\end{block}

\column{0.32\textwidth}
\begin{block}{\#2 Streaming Grounding}
Conditions change faster than queries arrive:\\
wind 2\,m/s at 08:00 $\to$ 7\,m/s at 14:00.\\[4pt]
\textit{Need: atomically consistent context at query receipt time.}
\end{block}

\column{0.32\textwidth}
\begin{block}{\#3 Faithfulness Gap}
LLM explanations can sound correct but not trace to actual computation (Lipton 2018).\\[4pt]
\textit{Need: explanations verifiable by agronomists \& EU AI Act audit.}
\end{block}
\end{columns}

\bigskip
\begin{center}
\small
\begin{tabular}{@{}clll@{}}
\toprule
\textbf{Challenge} & \textbf{State of the Art} & \textbf{Gap} & \textbf{My Contribution} \\
\midrule
Calibration & Softmax confidence, temperature scaling & Not distribution-free & C1: Conformal CIC \\
Streaming & Static KB retrieval & No IoT-stream grounding & C2: TSCA \\
Faithfulness & Post-hoc LLM explanation & Plausible $\neq$ faithful & C3: FAES \\
\bottomrule
\end{tabular}
\end{center}

\end{frame}

% SLIDE 5 — The Calibration Insight (C1)
% =====================================================================
\begin{frame}{The Calibration Insight: Conformal Prediction Sets $\to$ HITL Policy}

\begin{columns}[T]
\column{0.46\textwidth}
\textbf{Why not softmax confidence?}
\begin{itemize}\small
  \item Overconfident on OOD agricultural commands (Guo et al. 2017)
  \item No distribution-free guarantees
  \item Threshold-tuning is manual and brittle
\end{itemize}

\medskip
\textbf{Conformal prediction gives:}
\[
  \mathbb{P}\!\bigl(y \in \mathcal{C}(x)\bigr) \geq 1 - \alpha
\]
Distribution-free. One calibration forward pass. Deployable on Jetson Orin.

\column{0.52\textwidth}
\begin{center}
\small
\begin{tabular}{@{}p{3.0cm}lp{2.6cm}@{}}
\toprule
\textbf{Prediction Set} & \textbf{Meaning} & \textbf{HITL Decision} \\
\midrule
$\{\texttt{COMMAND}\}$ & High confidence & Proceed (safety check only) \\
$\{\texttt{COMMAND, ABORT}\}$ & Ambiguous & HITL mandatory \\
All 5 classes & OOD input & Reject + rephrase \\
\bottomrule
\end{tabular}
\end{center}

\medskip
\begin{exampleblock}{Research Novelty}
\small
HITL rate = tunable parameter (via $\alpha$). We empirically study the
\textbf{alert-fatigue vs.\ safety tradeoff} across $\alpha \in [0.01, 0.20]$.
\end{exampleblock}
\end{columns}

\medskip
\begin{alertblock}{Safety-Asymmetric Extension (Mondrian Conformal)}
\small
\texttt{ABORT} class gets stricter coverage ($\alpha_{\text{abort}} = 0.01$) than routine queries
($\alpha_{\text{query}} = 0.10$) --- per-class Mondrian conformal prediction.
\end{alertblock}

\end{frame}

% SLIDE 6 — HITL Design
% =====================================================================
\begin{frame}{Human-in-the-Loop Gate: Rule-Auditable, Not Learned}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{5 Rule-Auditable Trigger Conditions:}
\begin{enumerate}\small
  \item Action $\in$ \{\texttt{SPRAY, ABORT\_MISSION,}\\ \texttt{DOSAGE\_CHANGE, ZONE\_OVERRIDE}\}
  \item Conformal set non-singleton ($\geq$2 intents)
  \item Any entity unverified against KB
  \item User role = \texttt{OBSERVER} (read-only)
  \item Upstream data quality check failed
\end{enumerate}

\medskip
\small Each trigger $\to$ \textbf{explicit human-readable audit string}\\
(no LLM involved in trigger explanation)

\column{0.50\textwidth}
\begin{block}{Operator Response Options}
\begin{itemize}\small
  \item \texttt{YES} $\to$ execute command as proposed
  \item \texttt{NO} $\to$ abort; log refusal reason
  \item \texttt{MODIFY} $\to$ operator specifies override; system re-grounds
\end{itemize}
\end{block}

\medskip
\begin{block}{Why Rule-Auditable (Not Learned)?}
\small
A learned gate cannot be audited by an agronomist or regulator. \\
Rule-auditable gates are: version-controlled, explainable by design, testable with unit tests.
\end{block}

\end{columns}

\medskip
\begin{center}
\small
\textbf{HITL is not a limitation --- it is the research subject.} \\
RQ1 asks: what is the optimal $\alpha$ for each action class to minimise\\ 
unnecessary HITL while maintaining $\geq$0.99 recall on \texttt{ABORT}?
\end{center}

\end{frame}

\begin{frame}{Faithful Explanation (C3): Three Levels of Verifiability}

\begin{block}{The Core Distinction}
\textbf{Plausible explanation}: ``I recommended spraying because conditions were suitable.'' \\
\textbf{Faithful explanation}: ``I recommended spraying because: (1) soil moisture = 41\% $<$ 45\% threshold
[AGROVOC:soil-moisture-threshold]; (2) wind = 1.8\,m/s $<$ 3\,m/s spray limit [EPPO:spray-conditions];
(3) last spray was 6 days ago [audit-log:event-2847]; HITL triggered: action=SPRAY.''
\end{block}

\begin{columns}[T]
\column{0.32\textwidth}
\begin{exampleblock}{Level 1: Rule-Auditable}
\small
HITL trigger reasons are explicit rule-derived strings. No LLM involved.
\end{exampleblock}

\column{0.32\textwidth}
\begin{exampleblock}{Level 2: KB-Traceable}
\small
Every command element cites PPDB/EPPO/AGROVOC source URI as verifiable provenance.
\end{exampleblock}

\column{0.32\textwidth}
\begin{exampleblock}{Level 3: Attribution-Grounded}
\small
Integrated Gradients attribute which input tokens drove the LLM's intent prediction.
\end{exampleblock}
\end{columns}

\medskip
\begin{alertblock}{Connection to My BVF Research}
\small
BVF uses Fisher-Rao geometry to trace representation transitions across Llama/OLMo/Gemma/Mistral/Qwen.
C3 applies the same analytical lens to the agricultural fine-tuned LLM: which contextual features
(sensor readings, KB entities) activate each intent class in representation space.
\textbf{BVF $\to$ C3 is a direct research lineage, not a stretch.}
\end{alertblock}

\end{frame}

% SLIDE 8 — Metaflow as Scientific Infrastructure
% =====================================================================
\begin{frame}{Metaflow: Scientific Necessity, Not Engineering Preference}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Why Reproducibility Is a Scientific Problem Here}
\begin{itemize}\small
  \item Agricultural AI experiments are weather-dependent, season-bound, farm-specific
  \item Published results cannot be replicated without complete pipeline versioning
  \item Supervisors in \textbf{Lyon and Milan} must inspect \emph{identical} pipeline state
\end{itemize}

\medskip
\textbf{Metaflow's ``3 Vs'' (Tagliabue et al., MLSys 2023):}
\begin{itemize}\small
  \item \textbf{Velocity}: Iterate on C1 without re-running IoT ingestion
  \item \textbf{Validation}: Both supervisors inspect same artifact-level run
  \item \textbf{Versioning}: Model regression $\to$ detectable by run-ID comparison
\end{itemize}

\column{0.50\textwidth}
\begin{block}{Prototype $\to$ Production: Same Code}
\begin{center}
\small
\texttt{@batch(gpu=1)} on fine-tuning step:\\
--- Local: falls back to CPU\\
--- AWS: uses GPU endpoint\\[4pt]
\textbf{Pipeline validated in Lyon (M6--8)\\
deploys at UniMI farm in Milan (M20)\\
with zero code changes.}
\end{center}
\end{block}

\medskip
\begin{block}{DAG Concepts in This PhD}
\small
\textbf{Business Logic} = HITL rules (version-controlled)\\
\textbf{Environment} = @batch / @kubernetes / local\\
\textbf{Documentation} = @card $\to$ agronomist reports
\end{block}
\end{columns}

\end{frame}
